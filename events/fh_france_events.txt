  id = fh_france.70

  desc = fh_france.70.d
  picture = GFX_report_event_generic
  is_triggered_only = yes

  option = {
    name = fh_france.70.a
    if = {
      limit = { NOT = { has_idea = fh_fra_industrial_recovery_spirit } }

    if = {
      limit = { has_idea = fh_fra_industrial_recovery_spirit }
      add_political_power = 15
      add_stability = 0.01
    }
  }
}
